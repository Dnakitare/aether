{{- else if contains "NodePort" .Values.apiServer.service.type }}

The Aether API is available at:

  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "aether.fullname" . }}-api)
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT

{{- else if contains "LoadBalancer" .Values.apiServer.service.type }}

The Aether API is available at:

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "aether.fullname" . }}-api --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.apiServer.service.port }}

NOTE: It may take a few minutes for the LoadBalancer IP to be available.

{{- else if contains "ClusterIP" .Values.apiServer.service.type }}

The Aether API is running on ClusterIP. To access it:

  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "aether.name" . }},app.kubernetes.io/component=api-server" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:8080

Then access the API at: http://127.0.0.1:8080

{{- end }}
